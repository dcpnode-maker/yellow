function validateToml(text: string): string[] {
  const errors: string[] = [];
  let section = "";
  for (const [index, sourceLine] of text.split(/\r?\n/).entries()) {
    const line = sourceLine.replace(/#.*/, "").trim();
    if (!line) continue;
